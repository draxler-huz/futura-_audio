package com.example.musicalarm

import android.os.Bundle
import android.widget.Button
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity

class MainActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        // Test button to schedule an alarm 10 seconds from now
        val testButton = findViewById<Button>(R.id.btnTestAlarm)
        testButton.setOnClickListener {
            val triggerTime = System.currentTimeMillis() + 10_000L // 10 seconds
            val scheduler = AlarmScheduler(this)
            
            // 101 is a unique ID for this test alarm
            scheduler.schedule(triggerTime, alarmId = 101)
            
            Toast.makeText(this, "Alarm scheduled in 10 seconds!", Toast.LENGTH_SHORT).show()
        }
    }
}
